<div id="snow-card-frame-wrap" style="width:100%;overflow:hidden;">
  <iframe
    id="snow-card-frame"
    src="URL_NA_VASH_INDEX_HTML"
    style="width:100%;height:12000px;border:0;display:block;overflow:hidden;"
    scrolling="no"
    loading="eager"
    allowfullscreen>
  </iframe>
</div>
<script>
(function(){
  var iframe = document.getElementById('snow-card-frame');
  if (!iframe) return;

  function openTildaPopup(targets){
    var target = '#popup:tours';
    if (targets && targets.length) target = targets[0] || target;

    try {
      var link = document.createElement('a');
      link.href = target;
      link.style.position = 'absolute';
      link.style.left = '-9999px';
      link.textContent = 'open popup';
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    } catch(e) {}

    try {
      window.location.hash = target.replace('#', '');
      window.dispatchEvent(new HashChangeEvent('hashchange'));
    } catch(e) {}

    try {
      document.querySelectorAll('a[href="' + target + '"]').forEach(function(a){ a.click(); });
    } catch(e) {}
  }

  window.addEventListener('message', function(event){
    var data = event.data || {};
    if (data.type === 'snow-card-height' && data.height) {
      iframe.style.height = Math.ceil(Number(data.height)) + 'px';
    }
    if (data.type === 'open-snow-card-popup') {
      openTildaPopup(data.targets);
    }
  });
})();
</script>
